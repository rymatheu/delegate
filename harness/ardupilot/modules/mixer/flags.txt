# Compile flags of the mixer module, added to the shared ones in ardupilot/build.sh.
# AP_Motors uses APM_BUILD_TYPE(...), which needs the vehicle the library is built for; waf
# defines it per vehicle (Tools/ardupilotwaf/ardupilotwaf.py). It also selects air-density
# compensation (AP_MOTORS_DENSITY_COMP = 1 for ArduCopter) and the MOT_ parameter prefix.
-DAPM_BUILD_DIRECTORY=APM_BUILD_ArduCopter
# The SITL board's configure step defines AP_SIM_ENABLED (Tools/ardupilotwaf/boards.py, class sitl,
# cfg.define('AP_SIM_ENABLED', 1)) in waf's ap_config.h, which ardupilot/build.sh does not write;
# AP_AHRS.h uses it in an expression, so it must be defined.
-DAP_SIM_ENABLED=1
