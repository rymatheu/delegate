%  the delegate-port worktree), drawn as the cascade of the ported classes:
%  AC_WPNav.update_wpnav sets AC_PosControl's desired position, velocity
%  and acceleration; AC_PosControl.NE_update_controller and
%  D_update_controller produce the thrust vector, the heading (AutoYaw,
%  WP_YAW_BEHAVIOR 2: AC_PosControl's yaw from the desired velocity) and
%  the throttle (set_throttle_out); AC_AttitudeControl's
%  input_thrust_vector_heading stores the body-rate target; on the next
%  tick AC_AttitudeControl_Multi.rate_controller_run turns it into the roll,
%  pitch and yaw inputs of AP_MotorsMatrix, whose output is the four
%  actuators. Boxes with a double left edge keep state between ticks.

\begin{tikzpicture}[

  tick/.style={font=\scriptsize, text=fwMute, inner sep=1pt},
  frame/.style={draw=fwHair, line width=0.4pt},
  mod/.style={draw=fwInk!60, line width=0.45pt},
  ttl/.style={font=\footnotesize, text=fwInk, anchor=base, inner sep=0},

]
% ------------------------------------------------------------------
%  Geometry: top row y = 2.40 ... 3.45, bottom row y = 0.95 ... 1.90;
%  top modules x = 0.52 ... 8.20 (gaps 0.82, 0.78), bottom row
%  right to left: rate loop, motors, plant
% ------------------------------------------------------------------
\def\Ta{2.40} \def\Tb{3.45} \def\Tm{2.925}   % top row
\def\Ba{0.95} \def\Bb{1.90} \def\Bm{1.425}   % bottom row

% one module: #1 x-left, #2 x-right, #3 y-bottom, #4 y-top,
%             #5 class, #6 Lean function(s), #7 state note (empty = memoryless)
\newcommand{\module}[7]{%
  \draw[mod] (#1,#3) rectangle (#2,#4);
  \if\relax\detokenize{#7}\relax\def\cx{(#1+#2)/2}\else
    \def\cx{(#1+#2)/2+0.025}%
    \draw[mod] (#1+0.05,#3) -- (#1+0.05,#4);
    \node[mem] at ({\cx},#3+0.13) {#7};
  \fi
  \node[ttl] at ({\cx},#4-0.27) {#5};
  \node[fn]  at ({\cx},#4-0.36) {#6};}
% measured input: #1 x, #2 y on the frame, #3 +1 from above / -1 from below, #4 name
\newcommand{\inport}[4]{%
  \draw[tap] (#1,#2+#3*0.405) -- (#1,#2);
  \draw[port] (#1,#2+#3*0.45) circle (0.045);
  \node[tick, anchor=west] at (#1+0.05,#2+#3*0.45) {#4};}

% ================= modules ==========================================
\module{0.52}{2.04}{\Ta}{\Tb}{AC\_WPNav}{update\_wpnav}{path legs}
\module{2.86}{5.16}{\Ta}{\Tb}{AC\_PosControl}{NE\_update\_controller\\D\_update\_controller}{targets, PIDs}
\module{5.94}{8.20}{\Ta}{\Tb}{AC\_AttitudeControl}{input\_thrust\_vector\_\\heading}{attitude target}
\module{5.30}{8.20}{\Ba}{\Bb}{AC\_AttitudeControl\_Multi}{rate\_controller\_run}{rate loop, 3 PIDs}
\module{2.22}{4.58}{\Ba}{\Bb}{AP\_MotorsMatrix}{output}{throttle filter}

% ================= cascade ==========================================
\draw[sig] (2.04,\Tm) -- (2.86,\Tm)
  node[tick, midway, anchor=south, yshift=0.9pt, align=center, inner xsep=0pt] {desired\\pos, vel,\\acc};
\draw[sig] (5.16,\Tm) -- (5.94,\Tm)
  node[tick, midway, anchor=south, yshift=0.9pt, align=center, inner xsep=0pt] {thrust\\vector,\\heading};
\draw[sig] (7.07,\Ta) -- (7.07,\Bb)
  node[tick, midway, anchor=east, xshift=-0.5pt, align=right] {body-rate\\target};
\draw[sig] (5.30,\Bm) -- (4.58,\Bm)
  node[tick, midway, anchor=north, yshift=-0.3pt, align=center, inner xsep=0pt] {roll,\\pitch,\\yaw};
% throttle bypasses the attitude and rate loops (set_throttle_out)
\draw[->, draw=fwBlue!70, line width=0.45pt] (3.70,\Ta) -- (3.70,\Bb)
  node[tick, midway, anchor=west, text=fwBlue!70] {throttle};

% ================= measured inputs ==================================
\inport{0.90}{\Tb}{1}{pos, vel}
\inport{3.34}{\Tb}{1}{pos, vel, acc, dt}
\inport{7.07}{\Tb}{1}{att}
\inport{7.07}{\Ba}{-1}{rates, dt}

% ================= output ===========================================
\draw[frame] (0.40,\Bm-0.20) rectangle (1.30,\Bm+0.20);
\node[font=\footnotesize\scshape, text=fwMute, inner sep=0] at (0.85,\Bm) {plant};
\draw[sig] (2.22,\Bm) -- (1.30,\Bm)
  node[tick, midway, anchor=south, yshift=0.6pt, align=center, inner xsep=0pt] {actuators\\$m_1$--$m_4$};
