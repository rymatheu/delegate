% ==================================================================
%  Pipeline -- the whole procedure: ArduPilot's C++ control classes are
%  translated into a Lean model, the model is proved against a contract
%  (proof lane, top), the model is validated against the source by running
%  both (execution lane, bottom), and the checked result is carried back
%  to the source
%  one sigconf text width (8.2 cm), placed as a plain figure in one of the
%  two text areas, \input{figs/pipeline} at natural size (no \resizebox),
%  pdfLaTeX
%  Preamble:  \usepackage{tikz}  \usetikzlibrary{arrows.meta,calc}
%             (\mathbb and \checkmark come with acmart's newtxmath; outside
%             acmart add \usepackage{amssymb})
%  TikZ only, no image. Every label is a plain word; titles \footnotesize,
%  bodies and labels \scriptsize, nothing smaller; no code body is shown,
%  the boxes name the artifacts.
%  Translation row (middle, left to right): the C++ source (ArduPilot's
%  control classes, AC_*.cpp and AP_*.cpp, e.g. AC_AttitudeControl_Multi),
%  parsed by clang into the clang AST, translated by cpp2lean into the Lean
%  model (the same classes as .lean files in the delegate-port worktree,
%  Float32 arithmetic). The Lean model feeds both lanes.
%  Proof lane (top): the Lean model lifts to M_ctrl, the plant model (rigid
%  body over the reals, plant.tex) lifts to M_plant, the closed loop is
%  M = M_ctrl cap M_plant; with the contract (phi, psi), A from phi and G
%  from psi, the obligation M cap A subset G is checked in Lean.
%  Execution lane (bottom): the C++ compiled by clang and the Lean model
%  compiled by leanc are each run separately, in their own closed loop;
%  the two trajectories meet at the
%  numerical validation, executable model against source. The figure keeps
%  this box generic; the evidence behind it is oracle-diff.tex and the
%  determinism runs in sim/.
%  Transfer: the heavy arrow from the proof down into the C++ source,
%  "holds of the source"; the validation joins it through the dashed
%  connector up the left edge, "justifies".
%  Boxes: rounded, white body, title area fwHair at 40 percent; the four
%  primary boxes (C++ source, Lean model, proof, validation) are the
%  largest. Lanes tinted fwHair at 15 percent. Accents: ochre for phi and A,
%  blue for psi and G only.
%  The earlier full-width version is archive/pipeline-wide.tex.
% ==================================================================

% one stage: #1 name, #2 center x, #3 center y, #4 width, #5 height (cm),
%            #6 title, #7 body, #8 tag (empty = none); title area 0.36
\newcommand{\pipestage}[8]{%
  \node[minimum width=#4cm, minimum height=#5cm, inner sep=0, outer sep=0] (#1) at (#2,#3) {};
  \begin{scope}
    \clip[rounded corners=2pt] (#1.south west) rectangle (#1.north east);
    \fill[white] (#1.south west) rectangle (#1.north east);
    \fill[fwHair!40] ($(#1.north west)+(0,-0.36)$) rectangle (#1.north east);
  \end{scope}
  \draw[boxline] (#1.south west) rectangle (#1.north east);
  \node[fwcap, anchor=base west] at ($(#1.north west)+(0.14,-0.255)$) {#6};
  \node[body, anchor=north west] at ($(#1.north west)+(0.14,-0.47)$) {#7};
  \if\relax\detokenize{#8}\relax\else
    \node[tag, anchor=base east] at ($(#1.north east)+(-0.08,-0.255)$) {#8};\fi}
% binary icon (as in overview-D): #1 name, #2 lower-left corner
\newcommand{\pipebin}[2]{%
  \begin{scope}[shift={#2}]
    \draw[fill=white, draw=fwInk!70, line width=0.45pt] (0,0) rectangle (0.5,0.3);
    \draw[fwInk!45, line width=0.4pt] (0.08,0.22) -- (0.34,0.22);
    \draw[fwInk!45, line width=0.4pt] (0.08,0.15) -- (0.42,0.15);
    \draw[fwInk!45, line width=0.4pt] (0.08,0.08) -- (0.27,0.08);
    \coordinate (#1N) at (0.25,0.3);  \coordinate (#1S) at (0.25,0);
    \coordinate (#1W) at (0,0.15);    \coordinate (#1E) at (0.5,0.15);
  \end{scope}}
